Private m_Lineas() As LineaC31
Private m_nLin As Long

Private Sub UserForm_Initialize()
    Me.Caption = "SIC-MUNI - Contrataciones Menores"
    lblSesion.Caption = modSecurity.g_Nombre & "  [" & modSecurity.g_Rol & "]  DA " & modSecurity.g_DA
    txtDA.Text = modSecurity.g_DA
    cboTipo.AddItem "COMPRA": cboTipo.AddItem "SERVICIO": cboTipo.ListIndex = 0
    lstItems.ColumnCount = 6: lstItems.ColumnWidths = "30;70;190;50;50;60"
    lstCot.ColumnCount = 6: lstCot.ColumnWidths = "70;70;150;70;50;60"
    lstLineas.ColumnCount = 7: lstLineas.ColumnWidths = "40;40;40;40;40;60;70"
    m_nLin = 0
    AplicarRol
    MostrarPagina 1
End Sub

Private Sub UserForm_Activate()
    If Not modSecurity.SesionActiva() Then MsgBox "Sesion expirada.", vbExclamation: Unload Me
End Sub

Private Sub AplicarRol()
    ' Habilita botones segun permiso (la verificacion real la repite cada rutina de negocio).
    cmdCrearSolicitud.Enabled = PuedeHacer(P_SOLICITUD): cmdAgregarItem.Enabled = PuedeHacer(P_SOLICITUD)
    cmdEnviarCot.Enabled = PuedeHacer(P_SOLICITUD)
    cmdEvalCHB.Enabled = PuedeHacer(P_CHB): cmdDocExc.Enabled = PuedeHacer(P_CHB)
    cmdRegCot.Enabled = PuedeHacer(P_COTIZACION): cmdEvaluar.Enabled = PuedeHacer(P_COTIZACION)
    cmdCuadro.Enabled = PuedeHacer(P_COTIZACION): cmdAdjudicar.Enabled = PuedeHacer(P_ADJUDICAR)
    cmdEmitirC31.Enabled = PuedeHacer(P_C31): cmdAsociar.Enabled = PuedeHacer(P_C31)
    cmdRevParcial.Enabled = PuedeHacer(P_REVERSION): cmdRevTotal.Enabled = PuedeHacer(P_REVERSION)
    cmdAddLinea.Enabled = PuedeHacer(P_C31): cmdSaldo.Enabled = PuedeHacer(P_C31)
    cmdGenOrden.Enabled = PuedeHacer(P_ORDEN): cmdDocOrden.Enabled = PuedeHacer(P_ORDEN)
    cmdRecepcion.Enabled = PuedeHacer(P_ORDEN): cmdActa.Enabled = PuedeHacer(P_ORDEN)
    cmdAnularOrden.Enabled = PuedeHacer(P_ORDEN): cmdAnularSol.Enabled = PuedeHacer(P_SOLICITUD)
    cmdDocC31.Enabled = PuedeHacer(P_DOCS) And modSecurity.g_Rol <> ROL_US
End Sub

Private Sub MostrarPagina(ByVal n As Long)
    Dim c As MSForms.Control
    For Each c In Me.Controls
        If Len(c.Tag) > 0 Then c.Visible = (c.Tag = "P" & n)
    Next c
End Sub

Private Function SolActiva() As String
    SolActiva = Trim$(txtSol.Text)
    If Len(SolActiva) = 0 Then Err.Raise vbObjectError + 700, , "Cargue o cree una solicitud primero."
End Function

Private Sub Aviso(ByVal e As ErrObject)
    MsgBox e.Description, vbExclamation, "SIC-MUNI"
End Sub

EH: Aviso Err
End Sub

Private Sub Refrescar()
    Dim idSol As String, rS As Long, r As Long, sh As Worksheet
    idSol = SolActiva()
    rS = modProcurement.FilaSolicitud(idSol)
    Set sh = WS(SH_SOL)
    lblEstado.Caption = "Estado: " & GetV(sh, rS, "Estado") & "  |  Referencial Bs " & Format$(GetV(sh, rS, "MontoReferencial"), "#,##0.00") & _
                        "  |  " & GetV(sh, rS, "Modalidad")
    lblNroOrden.Caption = modPostAward.OrdenVigenteDeSolicitud(idSol)
    lstItems.Clear
    Set sh = WS(SH_DET)
    For r = 2 To LastRow(sh)
        If CStr(GetV(sh, r, "ID_Solicitud")) = idSol Then
            lstItems.AddItem GetV(sh, r, "Item")
            lstItems.List(lstItems.ListCount - 1, 1) = GetV(sh, r, "CodigoUNSPSC")
            lstItems.List(lstItems.ListCount - 1, 2) = GetV(sh, r, "Descripcion")
            lstItems.List(lstItems.ListCount - 1, 3) = GetV(sh, r, "Cantidad")
            lstItems.List(lstItems.ListCount - 1, 4) = Format$(GetV(sh, r, "PrecioRefTotal"), "#,##0.00")
            lstItems.List(lstItems.ListCount - 1, 5) = GetV(sh, r, "CumpleCHB")
        End If
    Next r
    lstCot.Clear
    Set sh = WS(SH_COT)
    For r = 2 To LastRow(sh)
        If CStr(GetV(sh, r, "ID_Solicitud")) = idSol Then
            lstCot.AddItem GetV(sh, r, "ID_Cotizacion")
            lstCot.List(lstCot.ListCount - 1, 1) = GetV(sh, r, "Proveedor_NIT")
            lstCot.List(lstCot.ListCount - 1, 2) = GetV(sh, r, "RazonSocial")
            lstCot.List(lstCot.ListCount - 1, 3) = Format$(GetV(sh, r, "MontoTotalCotizado"), "#,##0.00")
            lstCot.List(lstCot.ListCount - 1, 4) = GetV(sh, r, "CumplimientoTecnico")
            lstCot.List(lstCot.ListCount - 1, 5) = GetV(sh, r, "Recomendado")
        End If
    Next r
End Sub

Private Sub txtCodigo_AfterUpdate()
    ' Alerta CHB inmediata al digitar el codigo UNSPSC
    Dim res As CHBResultado
    res = modValidationCHB.ConsultarCHB(txtCodigo.Text)
    lblAlertaCHB.Caption = res.Mensaje
    lblAlertaCHB.ForeColor = IIf(res.EnCatalogo, RGB(192, 0, 0), RGB(0, 110, 0))
    If res.EnCatalogo Then
        If Len(txtDescItem.Text) = 0 Then txtDescItem.Text = res.Descripcion
        If Len(txtUnidad.Text) = 0 Then txtUnidad.Text = res.Unidad
    End If
End Sub

Private Sub cmdAgregarItem_Click()
    Dim a As String
    On Error GoTo EH
    If Not IsNumeric(txtCant.Text) Or Not IsNumeric(txtPrecio.Text) Then Err.Raise vbObjectError + 701, , "Cantidad y precio deben ser numericos."
    a = modProcurement.AgregarItem(SolActiva(), Trim$(txtCodigo.Text), txtDescItem.Text, CDbl(txtCant.Text), txtUnidad.Text, CDbl(txtPrecio.Text))
    If Len(a) > 0 Then MsgBox a & vbLf & vbLf & "Pase a la pestana 'Evaluacion CHB' para registrar compra por catalogo o la excepcion.", vbExclamation, "Alerta CHB"
    txtCodigo.Text = "": txtDescItem.Text = "": txtCant.Text = "": txtUnidad.Text = "": txtPrecio.Text = ""
    Refrescar
    Exit Sub
EH: Aviso Err
End Sub

'-------------------------- P4 Presupuesto / C-31 --------------------------
Private Function LeerLinea() As LineaC31
    Dim L As LineaC31
    If Not IsNumeric(txtImporte.Text) Then Err.Raise vbObjectError + 705, , "Importe invalido."
    L.Programa = Trim$(txtProg.Text): L.Proyecto = Trim$(txtProy.Text): L.ActObra = Trim$(txtAct.Text)
    L.Fuente = Trim$(txtFte.Text): L.Organismo = Trim$(txtOrg.Text): L.Partida = Trim$(txtPartida.Text)
    L.Importe = CDbl(txtImporte.Text)
    LeerLinea = L
End Function

Private Sub cmdSaldo_Click()
    Dim L As LineaC31, rS As Long
    On Error GoTo EH
    L = LeerLinea()
    rS = modProcurement.FilaSolicitud(SolActiva())
    lblSaldo.Caption = "Saldo disponible: Bs " & Format$(modBudgetEngine.SaldoDisponible(CStr(GetV(WS(SH_SOL), rS, "Cod_DA")), CStr(GetV(WS(SH_SOL), rS, "Cod_UE")), L), "#,##0.00")
    Exit Sub
EH: Aviso Err
End Sub

Private Sub cmdAddLinea_Click()
    On Error GoTo EH
    m_nLin = m_nLin + 1
    ReDim Preserve m_Lineas(1 To m_nLin)
    m_Lineas(m_nLin) = LeerLinea()
    With m_Lineas(m_nLin)
        lstLineas.AddItem .Programa
        lstLineas.List(lstLineas.ListCount - 1, 1) = .Proyecto: lstLineas.List(lstLineas.ListCount - 1, 2) = .ActObra
        lstLineas.List(lstLineas.ListCount - 1, 3) = .Fuente: lstLineas.List(lstLineas.ListCount - 1, 4) = .Organismo
        lstLineas.List(lstLineas.ListCount - 1, 5) = .Partida: lstLineas.List(lstLineas.ListCount - 1, 6) = Format$(.Importe, "#,##0.00")
    End With
    Exit Sub
EH: m_nLin = m_nLin - 1: Aviso Err
End Sub

Private Sub cmdEmitirC31_Click()
    Dim nro As String, msg As String
    On Error GoTo EH
    If m_nLin = 0 Then Err.Raise vbObjectError + 706, , "Agregue al menos una linea de imputacion."
    nro = modBudgetEngine.EmitirC31(SolActiva(), m_Lineas)
    txtNroC31.Text = nro
    lstLineas.Clear: m_nLin = 0
    MsgBox "Preventivo interno emitido: " & nro & vbLf & "Registre el C-31 en SIGEP y asocie su numero.", vbInformation
    Refrescar
    Exit Sub
EH:
    msg = Err.Description
    lstLineas.Clear: m_nLin = 0          ' evita duplicar lineas al reintentar
    MsgBox msg & vbLf & vbLf & "(Las lineas se limpiaron; vuelva a ingresarlas.)", vbExclamation, "SIC-MUNI"
End Sub
